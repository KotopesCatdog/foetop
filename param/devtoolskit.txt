// === ЭКСПОРТ ВСЕХ SELECTION KITS И ИХ СОДЕРЖИМОГО ===

const selectionKitsData = {};

// Проходим по всем SelectionKits из игры
for (const [kitId, kitData] of Object.entries(MainParser.SelectionKits || {})) {
    if (!kitData) continue;
    
    // Получаем опции выбора (с учётом разных эпох)
    let options = [];
    
    // Вариант 1: прямой массив options
    if (kitData.options && Array.isArray(kitData.options)) {
        options = kitData.options;
    }
    // Вариант 2: опции по эпохам
    else if (kitData.eraOptions) {
        // Берём опции для текущей эпохи или первой доступной
        const currentEraOptions = kitData.eraOptions[CurrentEra] || 
                                   Object.values(kitData.eraOptions)[0];
        if (currentEraOptions && currentEraOptions.options) {
            options = currentEraOptions.options;
        }
    }
    
    if (options.length === 0) continue;
    
    // Парсим каждый вариант выбора
    const choices = [];
    
    for (const option of options) {
        const item = option.item;
        if (!item) continue;
        
        const choice = {
            itemClass: item.__class__,
            amount: option.amount || 1
        };
        
        // Определяем тип и ID предмета
        if (item.__class__ === "BuildingItemPayload") {
            // Здание
            choice.type = "building";
            choice.id = item.cityEntityId;
            choice.name = MainParser.CityEntities[item.cityEntityId]?.name || item.cityEntityId;
            choice.icon = item.iconAssetName;
            
        } else if (item.__class__ === "UpgradeKitPayload") {
            // Апгрейд кит
            choice.type = "upgrade_kit";
            choice.id = item.upgradeItemId;
            choice.name = MainParser.BuildingUpgrades[item.upgradeItemId]?.upgradeItem?.name || item.upgradeItemId;
            choice.icon = item.iconAssetName;
            
        } else if (item.__class__ === "SelectionKitPayload") {
            // Вложенный набор выбора
            choice.type = "selection_kit";
            choice.id = item.selectionKitId;
            choice.name = MainParser.SelectionKits[item.selectionKitId]?.name || item.selectionKitId;
            choice.icon = item.iconAssetName;
            
        } else {
            // Другие типы (ресурсы, бусты и т.д.)
            choice.type = item.__class__;
            choice.id = item.id || item.cityEntityId || item.upgradeItemId;
            choice.name = item.name || "Неизвестный предмет";
        }
        
        choices.push(choice);
    }
    
    // Сохраняем данные о ките
    selectionKitsData[kitId] = {
        id: kitId,
        name: kitData.name,
        description: kitData.description || "",
        iconAssetName: kitData.iconAssetName,
        choices: choices,
        choicesCount: choices.length
    };
}

// Добавляем специальные случаи из Kits.specialCases
const specialCaseKits = {};
for (const [originalId, correctedId] of Object.entries(Kits.specialCases || {})) {
    if (selectionKitsData[correctedId]) {
        specialCaseKits[originalId] = selectionKitsData[correctedId];
    }
}

// Формируем итоговый объект
const result = {
    exportedAt: new Date().toISOString(),
    totalSelectionKits: Object.keys(selectionKitsData).length,
    selectionKits: selectionKitsData,
    specialCases: specialCaseKits
};

// Статистика по типам предметов в наборах
const typeStats = {};
for (const kit of Object.values(selectionKitsData)) {
    for (const choice of kit.choices) {
        typeStats[choice.type] = (typeStats[choice.type] || 0) + 1;
    }
}
result.typeStatistics = typeStats;

// Выводим результат
console.log('='.repeat(70));
console.log('📦 ЭКСПОРТ ВСЕХ SELECTION KITS');
console.log('='.repeat(70));
console.log(`✅ Всего наборов выбора: ${result.totalSelectionKits}`);
console.log(`📊 Статистика по типам:`);
for (const [type, count] of Object.entries(typeStats)) {
    console.log(`   - ${type}: ${count} предметов`);
}

// Копируем в буфер обмена
copy(JSON.stringify(result, null, 2));
console.log('📋 Данные скопированы в буфер обмена!');
console.log('='.repeat(70));

// Пример структуры первого кита

if (firstKit) {
    console.log('\n🔍 ПРИМЕР СТРУКТУРЫ (первый kit):');
    console.log(`ID: ${firstKit.id}`);
    console.log(`Название: ${firstKit.name}`);
    console.log(`Вариантов выбора: ${firstKit.choicesCount}`);
    console.log('Пример варианта:', firstKit.choices[0]);
}

// Дополнительно: группировка по типу для удобства
const groupedByType = {
    buildingKits: {},
    upgradeKits: {},
    selectionKits: {},
    otherKits: {}
};

for (const [kitId, kit] of Object.entries(selectionKitsData)) {
    const hasBuilding = kit.choices.some(c => c.type === 'building');
    const hasUpgrade = kit.choices.some(c => c.type === 'upgrade_kit');
    const hasSelection = kit.choices.some(c => c.type === 'selection_kit');
    
    if (hasBuilding && kit.choices.length === 1 && kit.choices[0].type === 'building') {
        groupedByType.buildingKits[kitId] = kit;
    } else if (hasUpgrade && kit.choices.length === 1 && kit.choices[0].type === 'upgrade_kit') {
        groupedByType.upgradeKits[kitId] = kit;
    } else if (hasSelection) {
        groupedByType.selectionKits[kitId] = kit;
    } else {
        groupedByType.otherKits[kitId] = kit;
    }
}

console.log('\n📂 ГРУППИРОВКА ПО ТИПАМ:');
console.log(`   - Простые наборы зданий: ${Object.keys(groupedByType.buildingKits).length}`);
console.log(`   - Простые наборы апгрейдов: ${Object.keys(groupedByType.upgradeKits).length}`);
console.log(`   - Наборы, содержащие другие наборы: ${Object.keys(groupedByType.selectionKits).length}`);
console.log(`   - Смешанные/другие: ${Object.keys(groupedByType.otherKits).length}`);

// Сохраняем группировку в отдельную переменную для копирования

    ...result,

};
